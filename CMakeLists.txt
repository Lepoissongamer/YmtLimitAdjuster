cmake_minimum_required(VERSION 3.20)
project(YmtLimitAdjuster VERSION 0.1.2 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)

if(BUILD_TESTING)
  add_executable(dependency_filter_tests tests/dependency_filter_tests.cpp)
  target_include_directories(dependency_filter_tests PRIVATE include)
  add_test(NAME dependency_filter COMMAND dependency_filter_tests)
  add_executable(pattern_tests tests/pattern_tests.cpp)
  target_include_directories(pattern_tests PRIVATE include)
  add_test(NAME patterns COMMAND pattern_tests)
  add_executable(reference_targets_tests tests/reference_targets_tests.cpp)
  target_include_directories(reference_targets_tests PRIVATE include)
  add_test(NAME reference_targets COMMAND reference_targets_tests)
  add_executable(memory_access_tests tests/memory_access_tests.cpp)
  target_include_directories(memory_access_tests PRIVATE include)
  add_test(NAME memory_access COMMAND memory_access_tests)
endif()

if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "GTA V Legacy requires a Windows x64 build.")
  endif()
  add_library(minhook STATIC
    third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
    third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
  target_include_directories(minhook PUBLIC third_party/minhook/include)
  add_library(YmtLimitAdjuster SHARED src/plugin.cpp src/game.cpp src/log.cpp)
  target_include_directories(YmtLimitAdjuster PRIVATE include src)
  target_compile_definitions(YmtLimitAdjuster PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(YmtLimitAdjuster PRIVATE minhook version user32)
  set_target_properties(YmtLimitAdjuster PROPERTIES PREFIX "" SUFFIX ".asi")
  if(MSVC)
    target_compile_options(YmtLimitAdjuster PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(YmtLimitAdjuster PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(YmtLimitAdjuster PRIVATE -static-libgcc -static-libstdc++)
  endif()
  install(TARGETS YmtLimitAdjuster RUNTIME DESTINATION . LIBRARY DESTINATION .)
  install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
  install(FILES docs/IMPLEMENTATION.md DESTINATION docs)
  install(FILES third_party/minhook/LICENSE.txt DESTINATION licenses RENAME MinHook.txt)
endif()
